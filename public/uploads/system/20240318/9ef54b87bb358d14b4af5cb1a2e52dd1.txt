<template>
  <el-checkbox
      v-model="checkAll"
      :indeterminate="isIndeterminate"
      @change="handleCheckAllChange"
  > 全选 </el-checkbox
  >
  <el-checkbox-group
      v-model="checkedName"
      @change="handleCheckedNameChange"
  >
    <el-checkbox
        v-for="name in props.updateplan"
        :key="name.mv_id"
        :label="name"
        :value="name.path"
    >{{ name.path }}</el-checkbox
    >
  </el-checkbox-group>
</template>

<script lang="ts" setup>
import { ref } from 'vue'
let props = defineProps({
  updateplan:{
    type:Object,
    default:{}
  }})
const checkAll = ref(false)
const isIndeterminate = ref(true)
const checkedName = ref()

const handleCheckAllChange = (val: boolean) => {
  console.log(props.updateplan);
  checkedName.value = val ? props.updateplan : []
  isIndeterminate.value = false

}
const handleCheckedNameChange = (value: string[]) => {
  const checkedCount = value.length
  checkAll.value = checkedCount === props.updateplan.length
  isIndeterminate.value = checkedCount > 0 && checkedCount < props.updateplan.length
}
</script>
